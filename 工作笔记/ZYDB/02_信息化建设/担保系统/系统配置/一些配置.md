logback-spring.xml配置日志输出便于排查问题

```
<!-- MyBatis SQL输出配置 -->
<!-- 输出Mapper接口的SQL语句 -->
<logger name="cn.com.csteamsun.manage.supervision.acceptance.mapper" level="DEBUG" />
<!-- 输出MyBatis的SQL参数 -->
<logger name="org.apache.ibatis.type.JdbcType" level="TRACE" />
<logger name="org.apache.ibatis.type.TypeHandler" level="TRACE" />
<logger name="java.sql.Statement" level="DEBUG" />
<logger name="java.sql.PreparedStatement" level="DEBUG" />
<logger name="java.sql.ResultSet" level="DEBUG" />
```
